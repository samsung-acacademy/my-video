/* ========================================================= "더 많은 질문 확인하기" 버튼 ========================================================= */ function toggleMoreFAQ() { /* 모든 FAQ 가져오기 */ const faqItems = document.querySelectorAll(".faq-item"); /* 더보기 버튼 가져오기 */ const moreButton = document.getElementById("faq-more-button"); /* FAQ 안의 문의 박스 가져오기 */ const contactArea = document.getElementById("contact-area"); /* 질문 4번이 현재 숨겨져 있는지 확인 숨겨져 있으면 → 펼치기 보이면 → 다시 접기 */ const isHidden = faqItems[3].classList.contains("hidden-faq"); /* ===================================================== 질문 4번부터 마지막 질문까지 처리 ===================================================== */ faqItems.forEach(function(item, index) { /* index 3 = 질문 4번 질문 4번 이후만 처리 */ if (index >= 3) { if (isHidden) { /* 현재 숨겨져 있다면 → hidden-faq 제거 → 질문을 화면에 표시 */ item.classList.remove("hidden-faq"); } else { /* 현재 보이고 있다면 → hidden-faq 추가 → 질문을 다시 숨김 */ item.classList.add("hidden-faq"); } } }); /* ===================================================== 펼쳐진 상태 / 접힌 상태에 따라 처리 ===================================================== */ if (isHidden) { // ========================= // 질문 펼치기 // ========================= // 문의 영역 표시 contactArea.classList.remove("hidden-contact"); // 버튼을 "질문 접기"로 변경 moreButton.innerHTML = '질문 접기 ↑'; // 모바일에서 버튼을 화면 아래에 고정 moreButton.classList.add("sticky-close"); } else { // ========================= // 질문 접기 // ========================= // 문의 영역 숨기기 contactArea.classList.add("hidden-contact"); // 버튼 고정 해제 moreButton.classList.remove("sticky-close"); // 버튼 내용을 원래대로 변경 moreButton.innerHTML = '더 많은 질문 확인하기' + ' (휴대폰번호 변경 · 교육비 결제 등) ↓'; } }
